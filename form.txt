
                            <form

                                  </label>
                                  <input
                                    name="village"
                                    type="text"
                                    defaultValue={
                                      userProfile
                                        ? `${userProfile.mandal || ""} / ${userProfile.district || ""}`
                                            .trim()

                                    }

                                      required
                                    />
                                  </div>

                                <div className="space-y-1.5 flex flex-col">
